// SCRIPT DE GOOGLE APPS SCRIPT // Pegá este código en script.google.com function doPost(e) { try { // Obtener los datos del formulario const data = JSON.parse(e.postData.contents); // Abrir o crear la hoja de cálculo const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName('Respuestas'); // Si no existe la hoja, crearla con los encabezados if (!sheet) { sheet = ss.insertSheet('Respuestas'); sheet.appendRow([ 'Timestamp', 'Curso', 'Antigüedad', 'Calidad Teórica', 'Calidad Vuelo', 'Instructores Claridad', 'Organización Programa', 'Estado Aeronaves', 'Demoras Técnicas', 'Obstáculo Coordinación', 'Comunicación', 'Cambio Principal', 'Motivación Elección', 'Servicios Nuevos', 'NPS', 'Motivo NPS', 'Puntaje General' ]); // Formatear encabezados const headerRange = sheet.getRange(1, 1, 1, 17); headerRange.setFontWeight('bold'); headerRange.setBackground('#2C3E50'); headerRange.setFontColor('#FFFFFF'); } // Agregar los datos a la hoja sheet.appendRow([ new Date(data.timestamp), data.curso || '', data.antiguedad || '', data.calidad_teorica || '', data.calidad_vuelo || '', data.instructores_claridad || '', data.organizacion_programa || '', data.estado_aeronaves || '', data.demoras_tecnicas || '', data.obstaculo_coordinacion || '', data.comunicacion || '', data.cambio_principal || '', data.motivacion_eleccion || '', data.servicios_nuevos || '', data.nps || '', data.nps_motivo || '', data.puntaje_general || '' ]); // Aplicar formato condicional a NPS const lastRow = sheet.getLastRow(); const npsCell = sheet.getRange(lastRow, 15); // Columna O (NPS) const npsValue = parseInt(data.nps); if (npsValue >= 0 && npsValue <= 6) { npsCell.setBackground('#E74C3C'); // Rojo para detractores npsCell.setFontColor('#FFFFFF'); } else if (npsValue >= 7 && npsValue <= 8) { npsCell.setBackground('#F39C12'); // Naranja para pasivos } else if (npsValue >= 9) { npsCell.setBackground('#27AE60'); // Verde para promotores npsCell.setFontColor('#FFFFFF'); } return ContentService.createTextOutput(JSON.stringify({ 'result': 'success', 'row': lastRow })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ 'result': 'error', 'error': error.toString() })).setMimeType(ContentService.MimeType.JSON); } } // Función para crear un dashboard automático function crearDashboard() { const ss = SpreadsheetApp.openById('1_OFd-YvpYWPLphtmVbVzaP1re3uUR6g9dxbjZ2CZR9w'); let dashboardSheet = ss.getSheetByName('Dashboard'); if (!dashboardSheet) { dashboardSheet = ss.insertSheet('Dashboard'); } else { dashboardSheet.clear(); } const sheet = ss.getSheetByName('Respuestas'); if (!sheet) { dashboardSheet.getRange('A1').setValue('No hay datos todavía. Esperando respuestas...'); return; } // Título dashboardSheet.getRange('A1').setValue('DASHBOARD - GÉNESIS AVIACIÓN'); dashboardSheet.getRange('A1:E1').merge(); dashboardSheet.getRange('A1').setFontSize(18).setFontWeight('bold').setBackground('#2C3E50').setFontColor('#FFFFFF'); // NPS Score const data = sheet.getDataRange().getValues(); const npsValues = data.slice(1).map(row => parseInt(row[14])).filter(val => !isNaN(val)); // Columna O (NPS) if (npsValues.length > 0) { const promotores = npsValues.filter(val => val >= 9).length; const pasivos = npsValues.filter(val => val >= 7 && val <= 8).length; const detractores = npsValues.filter(val => val <= 6).length; const npsScore = ((promotores - detractores) / npsValues.length * 100).toFixed(1); dashboardSheet.getRange('A3').setValue('NPS SCORE'); dashboardSheet.getRange('A3').setFontWeight('bold').setFontSize(14); dashboardSheet.getRange('B3').setValue(npsScore); dashboardSheet.getRange('B3').setFontSize(24).setFontWeight('bold'); if (npsScore >= 50) { dashboardSheet.getRange('B3').setFontColor('#27AE60'); } else if (npsScore >= 0) { dashboardSheet.getRange('B3').setFontColor('#F39C12'); } else { dashboardSheet.getRange('B3').setFontColor('#E74C3C'); } dashboardSheet.getRange('A4').setValue('Promotores (9-10):'); dashboardSheet.getRange('B4').setValue(promotores); dashboardSheet.getRange('A5').setValue('Pasivos (7-8):'); dashboardSheet.getRange('B5').setValue(pasivos); dashboardSheet.getRange('A6').setValue('Detractores (0-6):'); dashboardSheet.getRange('B6').setValue(detractores); // Promedios dashboardSheet.getRange('A8').setValue('PROMEDIOS DE CALIFICACIÓN'); dashboardSheet.getRange('A8').setFontWeight('bold').setFontSize(14); const calificaciones = [ ['Calidad Teórica', 3], ['Calidad Vuelo', 4], ['Organización', 6], ['Estado Aeronaves', 7], ['Comunicación', 10], ['Puntaje General', 16] ]; let row = 9; calificaciones.forEach(([nombre, col]) => { const valores = data.slice(1).map(r => parseInt(r[col])).filter(v => !isNaN(v)); const promedio = valores.length > 0 ? (valores.reduce((a, b) => a + b, 0) / valores.length).toFixed(2) : 'N/A'; dashboardSheet.getRange(`A${row}`).setValue(nombre + ':'); dashboardSheet.getRange(`B${row}`).setValue(promedio); row++; }); // Total respuestas dashboardSheet.getRange('A15').setValue('Total de Respuestas:'); dashboardSheet.getRange('A15').setFontWeight('bold'); dashboardSheet.getRange('B15').setValue(data.length - 1); } // Auto-ajustar columnas dashboardSheet.autoResizeColumns(1, 5); } // Función para ejecutar automáticamente el dashboard cada vez que llegue una respuesta function onEdit(e) { crearDashboard(); }
